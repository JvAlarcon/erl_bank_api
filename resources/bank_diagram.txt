// Paste the schema below in https://dbdiagram.io/d
Table customers {
  id SERIAL [pk, increment]
  name VARCHAR(60) [not null]
  created_at TIMESTAMP [pk, not null]
  updated_at TIMESTAMP [not null]
}

Table account_types {
  id SERIAL [pk, increment]
  label TEXT [not null]
}

Table accounts {
  id SERIAL [pk, increment]
  id_customer INTEGER [pk, not null, ref: > customers.id]
  id_type INTEGER [pk, not null, ref: > account_types.id]
  balance INTEGER [not null]
}

Table transaction_status {
  id SERIAL [pk, increment]
  label TEXT [not null]
}

Table transactions {
  id SERIAL [pk, increment]
  status_id INTEGER [not null, ref: > transaction_status.id]
  account_id INTEGER [not null, ref: > accounts.id]
  trace_id TEXT [not null, unique]
}

Table transfers_kind {
  label TEXT [pk, unique]
}

Table transfers {
  id SERIAL [pk, increment]
  transaction_id INTEGER [not null, ref: > transactions.id]
  kind TEXT [pk, not null, ref: > transfers_kind.label]
  trace_id INTEGER [not null, unique]
  from_account INTEGER [not null, unique]
  executed_at TIMESTAMP [pk, not null]
}

Table deposits {
  transfer_id INTEGER [pk, unique, not null, ref: > transfers.id]
  kind TEXT [not null, check: `kind = deposit`, 
    ref: > transfers_kind.label, ref: > transfers.kind]
  amount INTEGER [not null]
}

Table withdrawals {
  transfer_id INTEGER [pk, unique, not null, ref: > transfers.id]
  kind TEXT [not null, check: `kind = withdrawal`, 
    ref: > transfers_kind.label, ref: > transfers.kind]
  amount INTEGER [not null]
}

Table pix {
  transfer_id INTEGER [pk, unique, not null, ref: > transfers.id]
  kind TEXT [not null, check: `kind = pix`, 
    ref: > transfers_kind.label, ref: > transfers.kind]
  target_key TEXT [not null]
  amount INTEGER [not null]
}

Table credit {
  transfer_id INTEGER [pk, unique, not null, ref: > transfers.id]
  kind TEXT [not null, check: `kind = credit`, 
    ref: > transfers_kind.label, ref: > transfers.kind]
  pan TEXT [not null]
  amount INTEGER [not null]
}

Table debit {
  transfer_id INTEGER [pk, unique, not null, ref: > transfers.id]
  kind TEXT [not null, check: `kind = debit`, 
    ref: > transfers_kind.label, ref: > transfers.kind]
  pan TEXT [not null]
  amount INTEGER [not null]
}
